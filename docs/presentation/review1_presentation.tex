\begin{frame}{Feature Engineering or Dimensionality Reduction}
    \vspace{-0.15cm}
    \begin{beamerbox}
        \centering
        {\fontsize{8.5pt}{10pt}\selectfont \textbf{Dynamic Inflection Formulation:} \quad Velocity: $g_t = \frac{S_t - S_{t-1}}{S_{t-1}}$ \quad\textbar\quad Acceleration: $a_t = g_t - g_{t-1}$ \quad\textbar\quad Target: $\text{Surge}_t \in \{0, 1\}$ (Top 15\% $g_t > 0$)}
    \end{beamerbox}
    \vspace{0.15cm}

    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{beamerbox}
                {\fontsize{9pt}{11pt}\selectfont \textbf{\color{NavyPrimary}Engineered Predictive Features}}\\[0.1cm]
                \begin{itemize}\setlength{\itemsep}{2pt}
                    \item \textbf{1st-Order Growth ($g_t$):} Day-over-day search velocity.
                    \item \textbf{2nd-Order Acceleration ($a_t$):} Rate of velocity change.
                    \item \textbf{Temporal Lags ($t-1, t-2, t-3, t-7$):} Past signal states to enable leak-free forward forecasting.
                    \item \textbf{7-Day Rolling Metrics:} Rolling mean ($\mu_{7d}$) and volatility ($\sigma_{7d}$) capturing momentum shifts.
                    \item \textbf{Content Velocity:} 7-day upload count (\texttt{videos\_published\_7d}).
                \end{itemize}
            \end{beamerbox}
        \end{column}

        \begin{column}{0.48\textwidth}
            \begin{beamerbox}
                {\fontsize{9pt}{11pt}\selectfont \textbf{\color{NavyPrimary}Justification \& Measured Effect}}\\[0.1cm]
                \begin{itemize}\setlength{\itemsep}{2pt}
                    \item \textbf{Justification:} Raw volume alone fails to separate stable interest from breakout virality; 2nd-order acceleration captures the early exponential curve.
                    \item \textbf{Leakage Prevention:} Temporal lagging guarantees that model predictions use only historically available data.
                    \item \textbf{Measured Effect:} Produces clear separation of viral events; isolates 806 actionable surge instances with low noise.
                \end{itemize}
            \end{beamerbox}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Predictive Model Development}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{beamerbox}
                {\fontsize{9pt}{11pt}\selectfont \textbf{\color{NavyPrimary}Prediction Task \& Algorithms}}\\[0.1cm]
                \begin{itemize}\setlength{\itemsep}{2.5pt}
                    \item \textbf{Task:} Binary classification of next-window Demand Surge (\texttt{surge\_label} $\in \{0, 1\}$).
                    \item \textbf{Model 1 (Linear Baseline):} Logistic Regression with L2 regularization and balanced weights.
                    \item \textbf{Model 2 (Bagging):} Random Forest Classifier (100 estimators, max depth 6).
                    \item \textbf{Model 3 (Boosting):} HistGradientBoosting (GBDT) for non-linear interaction thresholds.
                \end{itemize}
            \end{beamerbox}
        \end{column}

        \begin{column}{0.48\textwidth}
            \begin{beamerbox}
                {\fontsize{9pt}{11pt}\selectfont \textbf{\color{NavyPrimary}Data Partitioning \& Settings}}\\[0.1cm]
                \begin{itemize}\setlength{\itemsep}{2.5pt}
                    \item \textbf{Strict Chronological 80/20 Split:}
                        \begin{itemize}\setlength{\itemsep}{1.5pt}
                            \item \textbf{Train:} Jan 15 -- Aug 12, 2026 ($N = 12,540$, Surges: 586)
                            \item \textbf{Test:} Aug 13 -- Oct 3, 2026 ($N = 3,180$, Surges: 208)
                        \end{itemize}
                    \item \textbf{Leakage Mitigation:} Chronological splitting ensures true out-of-sample forward evaluation.
                    \item \textbf{Imbalance Tuning:} \texttt{class\_weight='balanced'} penalizes false negatives on rare viral events (base rate: 4.7\%).
                \end{itemize}
            \end{beamerbox}
        \end{column}
    \end{columns}
\end{frame}

                    \item \textbf{YouTube Data API (3--10d Lead):} Captures influencer review velocity and comment sentiment consensus.
                    \item \textbf{Retailers \& OEMs:} Pre-allocate inventory 7--14 days early and synchronize marketing with viral momentum.
                \end{itemize}

\begin{frame}{Data Cleaning and Preprocessing}
    \begin{columns}[T]
        \begin{column}{0.49\textwidth}
            \begin{card}
                {\fontsize{8.5pt}{10pt}\selectfont \textbf{\color{NavyPrimary}Missing Values, Zeroes \& Deduplication}}\\[0.08cm]
                \begin{itemize}\setlength{\itemsep}{2pt}
                    \item \textbf{Zero Forward-Fill Policy:} Forward-filling prohibited to eliminate temporal lookahead bias and artificial trend continuity.
                    \item \textbf{Principled Zero Imputation:} Zero video uploads or sub-threshold search explicit as 0.0, keeping true zeroes distinct from missing.
                    \item \textbf{Query Version Control:} Tagged by registry version (\texttt{qv1--qv3}) to prevent duplicate fetches across collection cycles.
                    \item \textbf{Exact Title Matching:} Strict regex matching achieves $>80\%$ precision, eliminating noisy older device variants.
                \end{itemize}
            \end{card}
        \end{column}

        \begin{column}{0.49\textwidth}
            \begin{card}
                {\fontsize{8.5pt}{10pt}\selectfont \textbf{\color{NavyPrimary}Encoding, Scaling \& Verification Evidence}}\\[0.08cm]
                \begin{itemize}\setlength{\itemsep}{2pt}
                    \item \textbf{One-Hot Encoding:} Applied to \texttt{brand} (12 dummies) and \texttt{product\_type} (2 dummies) with \texttt{drop\_first=True}.
                    \item \textbf{StandardScaler:} Normalization applied to continuous features; identifiers and \texttt{surge\_label} left unscaled.
                    \item \textbf{Videos Table:} 3,600 raw $\rightarrow$ 3,500 clean (100 outdated query rows dropped).
                    \item \textbf{Comments Table:} 33,217 raw $\rightarrow$ 32,215 clean (empty/bot text purged).
                    \item \textbf{Audit Evidence:} 100\% verified in \texttt{cleaning\_report\_youtube.csv}.
                \end{itemize}

\begin{frame}{Feature Engineering or Dimensionality Reduction}
    \vspace{-0.1cm}
    \begin{card}
        \centering
        {\fontsize{8pt}{9.5pt}\selectfont \textbf{Dynamic Inflection Formulation:} \quad Velocity: $g_t = \frac{S_t - S_{t-1}}{S_{t-1}}$ \quad\textbar\quad Acceleration: $a_t = g_t - g_{t-1}$ \quad\textbar\quad Target: $\text{Surge}_t \in \{0, 1\}$ (Top 15\% $g_t > 0$)}
    \end{card}
    \vspace{0.15cm}

    \begin{columns}[T]
        \begin{column}{0.49\textwidth}
            \begin{card}
                {\fontsize{8.5pt}{10pt}\selectfont \textbf{\color{NavyPrimary}Engineered Features \& Temporal Lags}}\\[0.08cm]
                \begin{itemize}\setlength{\itemsep}{2pt}
                    \item \textbf{1st-Order Growth ($g_t$):} Day-over-day search velocity.
                    \item \textbf{2nd-Order Acceleration ($a_t$):} Rate of velocity change.
                    \item \textbf{Temporal Lags ($t-1, t-2, t-3, t-7$):} Past signal states enabling strictly leak-free forward forecasting.
                    \item \textbf{7-Day Rolling Metrics:} Rolling mean ($\mu_{7d}$) and volatility ($\sigma_{7d}$) capturing momentum shifts.
                    \item \textbf{Content Velocity:} 7-day upload count (\texttt{videos\_published\_7d}).
                \end{itemize}
            \end{card}
        \end{column}

        \begin{column}{0.49\textwidth}
            \begin{card}
                {\fontsize{8.5pt}{10pt}\selectfont \textbf{\color{NavyPrimary}Justification \& Measured Effect}}\\[0.08cm]
                \begin{itemize}\setlength{\itemsep}{2pt}
                    \item \textbf{Justification:} Raw volume alone cannot separate steady demand from sudden viral surges; 2nd-order acceleration captures the early exponential inflection.
                    \item \textbf{Leakage Prevention:} Temporal lagging guarantees that model predictions use only historically available observations.
                    \item \textbf{Measured Effect:} Produces clear bimodal separation; isolates 806 actionable surge events with minimal background noise.
                \end{itemize}
            \end{card}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Predictive Model Development}
    \begin{columns}[T]
        \begin{column}{0.49\textwidth}
            \begin{card}
                {\fontsize{8.5pt}{10pt}\selectfont \textbf{\color{NavyPrimary}Prediction Task \& Algorithms}}\\[0.08cm]
                \begin{itemize}\setlength{\itemsep}{2pt}
                    \item \textbf{Task:} Binary classification of next-window Demand Surge (\texttt{surge\_label} $\in \{0, 1\}$).
                    \item \textbf{Model 1 (Linear Baseline):} Logistic Regression with L2 regularization and balanced class weights.
                    \item \textbf{Model 2 (Bagging):} Random Forest Classifier (100 estimators, max depth 6 to prevent overfitting).
                    \item \textbf{Model 3 (Boosting):} HistGradientBoosting (GBDT) for non-linear feature interaction thresholds.
                \end{itemize}
            \end{card}
        \end{column}

        \begin{column}{0.49\textwidth}
            \begin{card}
                {\fontsize{8.5pt}{10pt}\selectfont \textbf{\color{NavyPrimary}Data Partitioning \& Protocol}}\\[0.08cm]
                \begin{itemize}\setlength{\itemsep}{2pt}
                    \item \textbf{Strict Chronological 80/20 Split:}
                        \begin{itemize}\setlength{\itemsep}{1pt}
                            \item \textbf{Train:} Jan 15 -- Aug 12, 2026 ($N = 12,540$, Surges: 586)
                            \item \textbf{Test:} Aug 13 -- Oct 3, 2026 ($N = 3,180$, Surges: 208)
                        \end{itemize}
                    \item \textbf{Leakage Mitigation:} Chronological splitting ensures true forward-looking evaluation on unseen future dates.
                    \item \textbf{Imbalance Tuning:} \texttt{class\_weight='balanced'} penalizes false negatives on rare viral surges (4.7\% base rate).
                \end{itemize}
            \end{card}
        \end{column}
    \end{columns}
\end{frame}
